Cryptographic hash function that computes a 256-bit digest using the SHA-2 algorithm. 

// Rust API
use sha256::Sha256;

let mut hasher = Sha256::new();
hasher.update(b"hello world");
let digest = hasher.finalize();
assert_eq!(digest.len(), 32);

// C API
#include <stdint.h>
#include <string.h>

struct Sha256;

extern void sha256_init(struct Sha256 *ctx);
extern void sha256_update(struct Sha256 *ctx, const uint8_t *data, size_t len);
extern void sha256_finalize(struct Sha256 *ctx, uint8_t out[32]);

void compute() {
    struct Sha256 ctx;
    sha256_init(&ctx);
    sha256_update(&ctx, (const uint8_t*)"abc", 3);
    uint8_t digest[32];
    sha256_finalize(&ctx, digest);
}
